git init - инициализация пустого репоизтория
git status - проверка состояния репозитория
git add - добавление файла в репозиторий
git commit - слохранение изменений файлов в репоизтории
git checkout -b - создание новой ветки и переход сразу же в нее
git switch <ветка> - переход на другую ветку
git merge - слияние веток
git branch - кол-во веток и их название
git branch <имя ветки> - создание ветки
